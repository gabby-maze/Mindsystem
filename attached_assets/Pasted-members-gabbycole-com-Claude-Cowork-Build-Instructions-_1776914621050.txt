Create these tables:

sql-- Family accounts
CREATE TABLE families (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email TEXT UNIQUE NOT NULL,
  family_name TEXT,
  tier TEXT NOT NULL DEFAULT 'free',
  created_at TIMESTAMP DEFAULT now(),
  ghl_contact_id TEXT,
  terms_agreed BOOLEAN DEFAULT false,
  terms_agreed_at TIMESTAMP
);

-- Progress tracking
CREATE TABLE progress (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  family_id UUID REFERENCES families(id),
  course_id TEXT NOT NULL,
  lesson_id INTEGER NOT NULL,
  completed_at TIMESTAMP DEFAULT now(),
  UNIQUE(family_id, course_id, lesson_id)
);

-- Comments
CREATE TABLE comments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  family_id UUID REFERENCES families(id),
  course_id TEXT NOT NULL,
  lesson_id INTEGER NOT NULL,
  comment_text TEXT NOT NULL,
  is_gabby_reply BOOLEAN DEFAULT false,
  created_at TIMESTAMP DEFAULT now()
);

-- Questions
CREATE TABLE questions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  family_id UUID REFERENCES families(id),
  course_id TEXT,
  question_text TEXT NOT NULL,
  is_answered BOOLEAN DEFAULT false,
  created_at TIMESTAMP DEFAULT now()
);

-- Notifications
CREATE TABLE notifications (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tier TEXT, -- null means all tiers
  title TEXT NOT NULL,
  message TEXT NOT NULL,
  link TEXT,
  created_at TIMESTAMP DEFAULT now()
);

-- Notification reads
CREATE TABLE notification_reads (
  family_id UUID REFERENCES families(id),
  notification_id UUID REFERENCES notifications(id),
  read_at TIMESTAMP DEFAULT now(),
  PRIMARY KEY (family_id, notification_id)
);

-- Maze model completion (for parent training unlock)
CREATE TABLE maze_completions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  family_id UUID REFERENCES families(id),
  completed_at TIMESTAMP DEFAULT now(),
  week_completed INTEGER
);

javascriptasync function syncToGHL(familyData) {
  await fetch('[GHL-WEBHOOK-URL]', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      event: 'member_activity',
      email: familyData.email,
      family_name: familyData.family_name,
      tier: familyData.tier,
      last_login: new Date().toISOString()

javascriptfunction getWeekProgress(enrollmentDate) {
  const now = new Date();
  const start = new Date(enrollmentDate);
  const daysDiff = Math.floor((now - start) / (1000 * 60 * 60 * 24));
  const weekNumber = Math.min(Math.floor(daysDiff / 7) + 1, 12);
  const daysRemaining = Math.max(84 - daysDiff, 0);
  return { weekNumber, daysRemaining };
}

javascriptasync function loadNotifications() {
  const { data } = await supabase
    .from('notifications')
    .select('*, notification_reads(*)')
    .or(`tier.eq.${getCurrentTier()},tier.is.null`)
    .order('created_at', { ascending: false })
    .limit(10);

  const unread = data.filter(n => n.notification_reads.length === 0);
  updateBellBadge(unread.length);
  renderNotifications(data);
}
